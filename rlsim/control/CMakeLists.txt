cmake_minimum_required(VERSION 3.10)
project(rlsim_control LANGUAGES CXX)

find_package(Torch REQUIRED)
find_package(TBB REQUIRED)
find_package(yaml-cpp REQUIRED)
if(NOT TARGET rlsim_core)
    add_subdirectory(../inference "${CMAKE_CURRENT_BINARY_DIR}/inference")
endif()
if(TARGET yaml-cpp::yaml-cpp)
    set(RLSIM_CONTROL_YAML yaml-cpp::yaml-cpp)
else()
    set(RLSIM_CONTROL_YAML yaml-cpp)
endif()
add_library(rlsim_control STATIC src/runtime.cpp src/policy_bridge.cpp src/robot_runtime.cpp)
get_filename_component(RLSIM_CONTROL_WORKSPACE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
target_compile_definitions(rlsim_control PRIVATE RLSIM_WORKSPACE_ROOT="${RLSIM_CONTROL_WORKSPACE}")
target_compile_features(rlsim_control PUBLIC cxx_std_17)
target_include_directories(rlsim_control PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
target_link_libraries(rlsim_control PUBLIC rlsim_core "${TORCH_LIBRARIES}" TBB::tbb ${RLSIM_CONTROL_YAML})
set_target_properties(rlsim_control PROPERTIES POSITION_INDEPENDENT_CODE ON)
