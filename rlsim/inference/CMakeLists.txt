cmake_minimum_required(VERSION 3.10)
project(rlsim_core LANGUAGES CXX)

set(ONNXRUNTIME_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../third/onnxruntime-linux-x64-1.20.1"
    CACHE PATH "ONNX Runtime installation")
find_path(RLSIM_ORT_INCLUDE onnxruntime_cxx_api.h PATHS "${ONNXRUNTIME_ROOT}/include" NO_DEFAULT_PATH)
find_library(RLSIM_ORT_LIBRARY onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH)
if(NOT RLSIM_ORT_INCLUDE OR NOT RLSIM_ORT_LIBRARY)
    message(FATAL_ERROR "Set ONNXRUNTIME_ROOT to a valid ONNX Runtime installation")
endif()
find_package(yaml-cpp REQUIRED)
if(TARGET yaml-cpp::yaml-cpp)
    set(RLSIM_YAML_TARGET yaml-cpp::yaml-cpp)
else()
    set(RLSIM_YAML_TARGET yaml-cpp)
endif()

add_library(rlsim_core src/model.cpp src/policy.cpp src/config.cpp)
target_compile_features(rlsim_core PUBLIC cxx_std_17)
target_include_directories(rlsim_core PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
target_include_directories(rlsim_core SYSTEM PRIVATE "${RLSIM_ORT_INCLUDE}")
target_link_libraries(rlsim_core PRIVATE "${RLSIM_ORT_LIBRARY}" ${RLSIM_YAML_TARGET})
target_compile_options(rlsim_core PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(rlsim_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

option(RLSIM_BUILD_TESTS "Build core offline tests and tools" OFF)
option(RLSIM_BUILD_TOOLS "Build the standalone policy validator and runner" ON)
if(RLSIM_BUILD_TOOLS OR RLSIM_BUILD_TESTS)
    add_executable(rlsim_policy_run ../../tools/policy_run.cpp)
    target_link_libraries(rlsim_policy_run PRIVATE rlsim_core)
endif()
if(RLSIM_BUILD_TESTS)
    enable_testing()
    add_executable(rlsim_core_probe ../../tools/model_probe/main.cpp)
    target_compile_definitions(rlsim_core_probe PRIVATE RLSIM_PROBE_CORE)
    target_include_directories(rlsim_core_probe SYSTEM PRIVATE "${RLSIM_ORT_INCLUDE}")
    target_link_libraries(rlsim_core_probe PRIVATE rlsim_core "${RLSIM_ORT_LIBRARY}")
    add_executable(policy_core_test ../../tools/tests/policy_core_test.cpp)
    target_include_directories(policy_core_test PRIVATE ../interfaces)
    target_link_libraries(policy_core_test PRIVATE rlsim_core)
    add_test(NAME policy_core COMMAND policy_core_test)
    find_package(Threads REQUIRED)
    add_executable(loop_lifecycle_test ../../tools/tests/loop_lifecycle_test.cpp)
    target_include_directories(loop_lifecycle_test PRIVATE ../control/include/rlsim)
    target_link_libraries(loop_lifecycle_test PRIVATE Threads::Threads)
    add_test(NAME loop_lifecycle COMMAND loop_lifecycle_test)
endif()
