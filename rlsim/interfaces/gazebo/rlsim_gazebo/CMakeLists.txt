cmake_minimum_required(VERSION 3.10)
project(rlsim_gazebo VERSION 3.0.0 LANGUAGES CXX)
get_filename_component(RLSIM_PACKAGE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
get_filename_component(RLSIM_WORKSPACE "${RLSIM_PACKAGE_SOURCE}/../../../.." ABSOLUTE)
find_package(catkin REQUIRED COMPONENTS roscpp std_msgs std_srvs sensor_msgs geometry_msgs gazebo_msgs robot_msgs robot_controller)
find_package(Threads REQUIRED)
find_package(gazebo REQUIRED)
find_package(Torch REQUIRED)
catkin_package(CATKIN_DEPENDS roscpp std_msgs std_srvs sensor_msgs geometry_msgs gazebo_msgs robot_msgs robot_controller)
add_subdirectory("${RLSIM_WORKSPACE}/rlsim/inference" "${CMAKE_CURRENT_BINARY_DIR}/inference")
add_subdirectory("${RLSIM_WORKSPACE}/rlsim/control" "${CMAKE_CURRENT_BINARY_DIR}/control")
add_executable(rlsim_gazebo_node src/main.cpp src/gazebo_node.cpp)
target_include_directories(rlsim_gazebo_node PRIVATE include "${RLSIM_WORKSPACE}/rlsim/interfaces" ${catkin_INCLUDE_DIRS})
target_compile_definitions(rlsim_gazebo_node PRIVATE RLSIM_WORKSPACE_ROOT="${RLSIM_WORKSPACE}")
target_link_libraries(rlsim_gazebo_node PRIVATE rlsim_control Threads::Threads ${catkin_LIBRARIES})
add_dependencies(rlsim_gazebo_node ${catkin_EXPORTED_TARGETS})
add_library(rlsim_web_camera_pose SHARED src/web_camera_pose.cpp)
target_compile_features(rlsim_web_camera_pose PRIVATE cxx_std_17)
target_include_directories(rlsim_web_camera_pose PRIVATE ${GAZEBO_INCLUDE_DIRS} ${catkin_INCLUDE_DIRS})
target_link_libraries(rlsim_web_camera_pose PRIVATE ${GAZEBO_LIBRARIES} ${catkin_LIBRARIES})
add_dependencies(rlsim_web_camera_pose ${catkin_EXPORTED_TARGETS})
install(TARGETS rlsim_gazebo_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS rlsim_web_camera_pose LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY ../worlds robots models DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
