cmake_minimum_required(VERSION 3.10)
project(robot_controller VERSION 3.0.0 LANGUAGES CXX)
find_package(catkin REQUIRED COMPONENTS controller_interface hardware_interface pluginlib roscpp realtime_tools control_toolbox urdf robot_msgs)
catkin_package(INCLUDE_DIRS include LIBRARIES robot_controller CATKIN_DEPENDS controller_interface hardware_interface pluginlib roscpp realtime_tools control_toolbox urdf robot_msgs)
add_library(robot_controller src/robot_controller.cpp)
target_include_directories(robot_controller PUBLIC include ${catkin_INCLUDE_DIRS})
target_link_libraries(robot_controller ${catkin_LIBRARIES})
add_dependencies(robot_controller ${catkin_EXPORTED_TARGETS})
install(TARGETS robot_controller LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
install(FILES plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
